/** * WMLC5 Raffle Ticket Checklist — backend * Paste this into Extensions > Apps Script in your Google Sheet, * then deploy as a Web App (see instructions below). * * This script uses 3 sheet tabs as simple key-value stores: * Status -> ticket number : "sold" | "given" * Assignments -> row number : seller name * Entries -> sold ticket records (ticket, amount, time) * * It auto-creates these tabs the first time it runs. */ function getSheet(name, headers) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); } return sheet; } function readAllRows(sheet) { var data = sheet.getDataRange().getValues(); data.shift(); // remove header row return data; } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // Handles GET requests: ?action=getAll function doGet(e) { var action = e.parameter.action || 'getAll'; if (action === 'getAll') { var statusSheet = getSheet('Status', ['ticket', 'state']); var assignSheet = getSheet('Assignments', ['row', 'name']); var entriesSheet = getSheet('Entries', ['ticket', 'amount', 'time']); var status = {}; readAllRows(statusSheet).forEach(function(r) { if (r[0] !== '') status[String(r[0])] = r[1]; }); var assignments = {}; readAllRows(assignSheet).forEach(function(r) { if (r[0] !== '') assignments[String(r[0])] = r[1]; }); var entries = []; readAllRows(entriesSheet).forEach(function(r) { if (r[0] !== '') entries.push({ ticket: String(r[0]), amount: r[1], time: r[2] }); }); return jsonResponse({ ok: true, status: status, assignments: assignments, entries: entries }); } return jsonResponse({ ok: false, error: 'unknown action' }); } // Handles POST requests: body = { action, ...payload } function doPost(e) { var body = JSON.parse(e.postData.contents); var action = body.action; var lock = LockService.getScriptLock(); lock.waitLock(10000); // prevent two simultaneous writes from clobbering each other try { if (action === 'setTicket') { // payload: { ticket: "0001", state: "sold" | "given" | null } var statusSheet = getSheet('Status', ['ticket', 'state']); setKeyValue(statusSheet, String(body.ticket), body.state); return jsonResponse({ ok: true }); } if (action === 'setTicketsBulk') { // payload: { tickets: ["0001","0002",...], state: "sold" | "given" | null } var statusSheet2 = getSheet('Status', ['ticket', 'state']); body.tickets.forEach(function(t) { setKeyValue(statusSheet2, String(t), body.state); }); return jsonResponse({ ok: true }); } if (action === 'setAssignment') { // payload: { row: "3", name: "Jane Doe" | null } var assignSheet = getSheet('Assignments', ['row', 'name']); setKeyValue(assignSheet, String(body.row), body.name); return jsonResponse({ ok: true }); } if (action === 'addEntry') { // payload: { ticket: "0001", amount: 10, time: 169999999 } var entriesSheet = getSheet('Entries', ['ticket', 'amount', 'time']); var existing = readAllRows(entriesSheet); var already = existing.some(function(r) { return keysMatch(r[0], String(body.ticket)); }); if (!already) { entriesSheet.appendRow(["'" + body.ticket, body.amount, body.time]); } return jsonResponse({ ok: true }); } if (action === 'removeEntry') { // payload: { ticket: "0001" } var entriesSheet2 = getSheet('Entries', ['ticket', 'amount', 'time']); removeRowByKey(entriesSheet2, String(body.ticket)); return jsonResponse({ ok: true }); } return jsonResponse({ ok: false, error: 'unknown action: ' + action }); } catch (err) { return jsonResponse({ ok: false, error: err.message }); } finally { lock.releaseLock(); } } // Normalizes a sheet cell value for comparison against a string key. // Google Sheets often strips leading zeros from numeric-looking text // (e.g. "0007" becomes the number 7), so we compare on the numeric // value when possible, falling back to a plain string compare. function keysMatch(cellValue, key) { var cellStr = String(cellValue); if (cellStr === key) return true; var cellNum = Number(cellValue); var keyNum = Number(key); if (!isNaN(cellNum) && !isNaN(keyNum) && cellNum === keyNum) return true; return false; } // Sets or deletes a key/value pair in a 2-column sheet (col A = key, col B = value) function setKeyValue(sheet, key, value) { var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (keysMatch(data[i][0], key)) { if (value === null || value === undefined) { sheet.deleteRow(i + 1); } else { sheet.getRange(i + 1, 2).setValue(value); } return; } } if (value !== null && value !== undefined) { // Force the key to be stored as literal text so it round-trips // exactly (preserves leading zeros, avoids number/string mismatches). sheet.appendRow(["'" + key, value]); } } // Removes a row where column A matches key (for Entries sheet) function removeRowByKey(sheet, key) { var data = sheet.getDataRange().getValues(); for (var i = data.length - 1; i >= 1; i--) { if (keysMatch(data[i][0], key)) { sheet.deleteRow(i + 1); } } }
